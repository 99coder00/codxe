// Xbox 360 zone code commands, applied after the PC commands of OpenAssetTools.

// GfxImage: the texture header follows the load def, pixel data is streamed after all
// assets into the large runtime block.
use GfxImage;
// The pixel data size is the card memory size (baseSize is not reliable in converted zones).
set count pixels cardMemory::platform[0];
// Pixels of images with delayLoadPixels are streamed after all assets, others inline.
set delayed pixels XFILE_BLOCK_LARGE_RUNTIME 4096 delayLoadPixels;
set block pixels XFILE_BLOCK_LARGE;
set allocalign pixels 4096;
// Inline pixels are streamed before the load def and texture header.
reorder:
    name
    pixels
    texture;

use GfxImageLoadDef;
set count texture 1;
// The texture header is the live D3D resource: it is kept in the virtual block.
set block texture XFILE_BLOCK_VIRTUAL;

// Pixel shaders are assets on console.
asset MaterialPixelShader AssetPixelShader;
use MaterialPixelShader;
set block XFILE_BLOCK_TEMP;

use GfxPixelShaderLoadDef;
set count cachedPart cachedPartSize / 4;
set count physicalPart physicalPartSize / 4;
set block physicalPart XFILE_BLOCK_PHYSICAL;
set allocalign physicalPart 32;

use GfxVertexShaderLoadDef;
set count cachedPart cachedPartSize / 4;
set count physicalPart physicalPartSize / 4;
set block physicalPart XFILE_BLOCK_PHYSICAL;
set allocalign physicalPart 32;

use MaterialPass;
set reusable vertexShaderArray;

use XAnimParts;
set count names boneCount[11];

// Console models carry high mip bounds for every surface of the first LOD.
use XModelStreamInfo;
set count highMipBounds XModel::lodInfo::numsurfs;

use snd_asset;
// XMA packets live in the large block (2 KB aligned for the XMA decoder)
set block data XFILE_BLOCK_LARGE;
set allocalign data 2048;
set count seekTable 1;

use XmaSeekTable360;
set arraysize entries count;

use GfxWorldStreamInfo360;
set count aabbTrees aabbTreeCount * 8;
set count leafRefs leafRefCount;
